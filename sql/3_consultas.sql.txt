
SELECT 
  l.titulo,
  a.nome AS autor,
  g.nome AS genero,
  l.ano_publicacao,
  l.quantidade_disponivel
FROM livros l
JOIN autores a ON l.id_autor = a.id_autor
JOIN generos g ON l.id_genero = g.id_genero;


SELECT 
  l.titulo,
  u.nome AS usuario,
  e.data_emprestimo,
  e.data_devolucao_prevista
FROM emprestimos e
JOIN livros l ON e.id_livro = l.id_livro
JOIN usuarios u ON e.id_usuario = u.id_usuario
WHERE e.status = 'ativo';


SELECT 
  u.nome,
  u.email,
  l.titulo,
  e.data_devolucao_prevista
FROM emprestimos e
JOIN usuarios u ON e.id_usuario = u.id_usuario
JOIN livros l ON e.id_livro = l.id_livro
WHERE e.data_devolucao_real IS NULL AND e.data_devolucao_prevista < CURRENT_DATE;


SELECT 
  u.nome,
  COUNT(e.id_emprestimo) AS total_emprestimos
FROM usuarios u
LEFT JOIN emprestimos e ON u.id_usuario = e.id_usuario
GROUP BY u.id_usuario
ORDER BY total_emprestimos DESC;


SELECT 
  l.titulo,
  COUNT(e.id_livro) AS vezes_emprestado
FROM emprestimos e
JOIN livros l ON e.id_livro = l.id_livro
GROUP BY l.id_livro
ORDER BY vezes_emprestado DESC
LIMIT 5;


SELECT 
  f.nome,
  f.cargo,
  COUNT(e.id_emprestimo) AS emprestimos_realizados
FROM funcionarios f
LEFT JOIN emprestimos e ON f.id_funcionario = e.id_funcionario
GROUP BY f.id_funcionario;
